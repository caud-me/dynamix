import { useRef, useState } from "react";

type Screen = "home" | "import" | "reviewer" | "setup" | "quiz";
type Result = "correct" | "incorrect" | null;

const answers = [
  "JavaFX was restricted exclusively to mobile and embedded architectures.",
  "JavaFX was redesigned as an independent, open-source project that evolves in parallel with the JDK.",
  "JavaFX was completely deprecated and replaced by Swing as the default GUI framework.",
  "JavaFX required native C++ compilation and could no longer run on JVM bytecode.",
];

function Icon({ children }: { children: React.ReactNode }) {
  return (
    <span className="icon" aria-hidden="true">
      {children}
    </span>
  );
}

function Row({
  icon,
  title,
  detail,
  active = false,
  onClick,
}: {
  icon: string;
  title: string;
  detail?: string;
  active?: boolean;
  onClick?: () => void;
}) {
  const Tag = onClick ? "button" : "div";
  return (
    <Tag className={`list-row${active ? " active" : ""}`} onClick={onClick}>
      <Icon>{icon}</Icon>
      <span className="row-copy">
        <span className="body-copy">{title}</span>
        {detail && <span className="eyebrow">{detail}</span>}
      </span>
    </Tag>
  );
}

export default function App() {
  const [screen, setScreen] = useState<Screen>("home");
  const [order, setOrder] = useState<"sequential" | "random">("sequential");
  const [includePast, setIncludePast] = useState(true);
  const [result, setResult] = useState<Result>(null);
  const [question, setQuestion] = useState(1);
  const fileRef = useRef<HTMLInputElement>(null);

  const startQuiz = () => {
    setResult(null);
    setQuestion(1);
    setScreen("quiz");
  };

  const continueQuiz = () => {
    setResult(null);
    setQuestion((value) => (value === 1 ? 2 : 1));
  };

  return (
    <main className="app-shell">
      {screen === "home" && (
        <section className="content page-content" aria-labelledby="home-title">
          <header className="heading-row">
            <h1 id="home-title">What are you cramming today?</h1>
            <button className="button compact" onClick={() => setScreen("import")}>
              <Icon>+</Icon> Add
            </button>
          </header>
          <div className="list">
            <p className="eyebrow section-label">Recent</p>
            <Row icon="□" title="JavaFX" detail="2h" active onClick={() => setScreen("reviewer")} />
            <Row icon="□" title="OOP" detail="2w" />
            <Row icon="□" title="Discrete Structures" detail="Nov 9" />
          </div>
        </section>
      )}

      {screen === "import" && (
        <section className="import-card" aria-labelledby="import-title">
          <div className="import-copy">
            <button className="text-action" onClick={() => setScreen("home")}>
              ‹ Cancel
            </button>
            <div>
              <h1 id="import-title">Add your reviewer</h1>
              <p>Select the .json generated by /whyeron-1.1f-catalyst skill here.</p>
            </div>
          </div>
          <button className="file-drop" onClick={() => fileRef.current?.click()}>
            <Icon>□</Icon>
            <strong>Click to select the file (.json)</strong>
            <span>Mainly /whyeron-1.1f-catalyst .json files</span>
          </button>
          <input
            ref={fileRef}
            className="visually-hidden"
            type="file"
            accept=".json,application/json"
            onChange={(event) => event.target.files?.length && setScreen("reviewer")}
          />
        </section>
      )}

      {screen === "reviewer" && (
        <section className="content page-content" aria-labelledby="reviewer-title">
          <button className="text-action" onClick={() => setScreen("home")}>
            ‹ Quit
          </button>
          <header className="heading-row">
            <div className="title-copy">
              <h1 id="reviewer-title">JavaFX</h1>
              <p>Short description here, maximum of 128 characters.</p>
            </div>
            <button className="button compact" onClick={startQuiz}>
              <Icon>✓</Icon> Start full exam
            </button>
          </header>
          <div className="list">
            <p className="eyebrow section-label">Sections</p>
            <Row
              icon="□"
              title="javaFX fundamentals"
              detail="27% confident"
              active
              onClick={() => setScreen("setup")}
            />
            <Row icon="✓" title="layout management" detail="mastered" />
            <Row icon="□" title="event-driven programming" />
          </div>
        </section>
      )}

      {screen === "setup" && (
        <section className="setup-card" aria-labelledby="setup-title">
          <Row icon="□" title="javaFX fundamentals" detail="27% confident" />
          <h1 id="setup-title">How do you like your questions ordered for this session?</h1>
          <div className="choice-list">
            <Row
              icon="▷"
              title="Sequential"
              detail="Good for trial and errors"
              active={order === "sequential"}
              onClick={() => setOrder("sequential")}
            />
            <Row
              icon="⌁"
              title="Randomized"
              detail="Good for mastering"
              active={order === "random"}
              onClick={() => setOrder("random")}
            />
          </div>
          <button className="list-row check-row" onClick={() => setIncludePast(!includePast)}>
            <span className={`checkbox${includePast ? " checked" : ""}`}>
              {includePast ? "✓" : ""}
            </span>
            <span className="row-copy">
              <span className="body-copy">Include past review cards?</span>
              <span className="eyebrow">This will add n extra cards from past sessions.</span>
            </span>
          </button>
          <div className="split-actions">
            <button className="button" onClick={() => setScreen("reviewer")}>
              Cancel
            </button>
            <button className="button" onClick={startQuiz}>
              Start
            </button>
          </div>
        </section>
      )}

      {screen === "quiz" && (
        <section className={`quiz-layout${question === 2 ? " with-code" : ""}`}>
          {question === 2 && (
            <aside className="code-card" aria-label="Code example">
              <code>{"public class main() {\n    ...\n}"}</code>
            </aside>
          )}
          <div className="content quiz-content">
            <p className="eyebrow">1 of 15 · Q1{result === "incorrect" ? "            12x" : ""}</p>
            <header className="question-row">
              <h1>
                What was the primary motivation for decoupling JavaFX from the standard Java
                Development Kit (JDK) starting with Java 11?
              </h1>
              <button className="button compact" onClick={() => setResult("correct")}>
                <Icon>⌒</Icon> Reveal
              </button>
            </header>
            <div className="answer-grid">
              {answers.map((answer, index) => {
                const selectedWrong = result === "incorrect" && index === 0;
                const muted = result !== null && !selectedWrong && index !== 0;
                return (
                  <button
                    className={`answer${selectedWrong ? " wrong" : ""}${index === 1 && result === "correct" ? " active" : ""}${muted ? " muted" : ""}`}
                    key={answer}
                    onClick={() => setResult(index === 1 ? "correct" : "incorrect")}
                    disabled={result !== null}
                  >
                    <span className="answer-number">{index + 1}</span>
                    <span>{answer}</span>
                  </button>
                );
              })}
            </div>
            {result && (
              <div className="feedback">
                <div>
                  <h2>{result === "correct" ? "That’s right" : "That’s not right"}</h2>
                  <p>
                    JavaFX unifies GUI components, modern 2D/3D graphics rendering, web views, and
                    multimedia (audio and video playback) within a single cohesive JavaFX API,
                    eliminating the fragmented libraries of earlier UI toolkits like AWT and Swing.
                  </p>
                </div>
                <button className="button compact" onClick={continueQuiz}>
                  Continue
                </button>
              </div>
            )}
            <button className="text-action" onClick={() => setScreen("reviewer")}>
              ‹ Quit
            </button>
          </div>
        </section>
      )}
    </main>
  );
}
